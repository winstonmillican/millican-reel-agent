import os
import uuid
import json
import subprocess
import httpx
import shutil
from pathlib import Path
from fastapi import FastAPI, HTTPException, BackgroundTasks, UploadFile, File
from fastapi.staticfiles import StaticFiles
from fastapi.middleware.cors import CORSMiddleware
from fastapi.responses import FileResponse
from pydantic import BaseModel
from typing import List, Optional

app = FastAPI(title="Millican Reel Agent")
app.add_middleware(CORSMiddleware, allow_origins=["*"], allow_methods=["*"], allow_headers=["*"])

UPLOAD_DIR = Path("/tmp/uploads")
OUTPUT_DIR = Path("/tmp/renders")
UPLOAD_DIR.mkdir(exist_ok=True)
OUTPUT_DIR.mkdir(exist_ok=True)

ANTHROPIC_API_KEY = os.environ.get("ANTHROPIC_API_KEY", "")
VIDEO_EXTENSIONS = {".mp4", ".mov", ".avi", ".mkv", ".m4v", ".webm"}

class ClipItem(BaseModel):
    id: str
    name: str
    local_path: str
    duration_estimate: Optional[str] = "unknown"

class GenerateRequest(BaseModel):
    brand_prompt: str
    clips: List[ClipItem]

class RenderRequest(BaseModel):
    clips: List[ClipItem]
    hook: str
    captions: List[str]
    cta: str
    job_id: Optional[str] = None

@app.get("/api/health")
def health():
    return {"status": "ok"}

@app.post("/api/upload")
async def upload_clips(files: List[UploadFile] = File(...)):
    session_id = str(uuid.uuid4())
    session_dir = UPLOAD_DIR / session_id
    session_dir.mkdir(exist_ok=True)
    clips = []
    for f in files:
        ext = Path(f.filename).suffix.lower()
        if ext not in VIDEO_EXTENSIONS:
            continue
        dest = session_dir / f.filename
        with dest.open("wb") as out:
            shutil.copyfileobj(f.file, out)
        duration = "unknown"
        try:
            r = subprocess.run(
                ["ffprobe", "-v", "error", "-show_entries", "format=duration",
                 "-of", "default=noprint_wrappers=1:nokey=1", str(dest)],
                capture_output=True, text=True, timeout=10)
            secs = float(r.stdout.strip())
            duration = f"{int(secs//60)}:{int(secs%60):02d}"
        except Exception:
            pass
        clips.append({"id": str(uuid.uuid4()), "name": f.filename,
                      "local_path": str(dest), "duration_estimate": duration})
    if not clips:
        raise HTTPException(400, "No valid video files found.")
    return {"clips": clips, "session_id": session_id}

@app.post("/api/generate")
async def generate_content(req: GenerateRequest):
    if not ANTHROPIC_API_KEY:
        raise HTTPException(500, "ANTHROPIC_API_KEY not set")
    clip_list = "\n".join([f"Clip {i+1}: {c.name} ({c.duration_estimate})" for i, c in enumerate(req.clips)])
    system = ("You are a social media content specialist for Millican Pecan Company, "
              "a family-owned pecan farm in San Saba, Texas. Brand voice: warm, Southern, authentic. "
              "Audience: women 40+. Output ONLY valid JSON — no markdown. "
              'Schema: {"hook":"string","cta":"string","captions":["string per clip"]}')
    user = (f"Brand brief: {req.brand_prompt}\n\nClips:\n{clip_list}\n\n"
            "hook: scroll-stopping opener under 10 words\n"
            "cta: call-to-action overlay under 8 words\n"
            "captions: one line per clip under 8 words\nReturn JSON only.")
    async with httpx.AsyncClient(timeout=30) as client:
        resp = await client.post("https://api.anthropic.com/v1/messages",
            headers={"x-api-key": ANTHROPIC_API_KEY, "anthropic-version": "2023-06-01",
                     "content-type": "application/json"},
            json={"model": "claude-sonnet-4-20250514", "max_tokens": 1000,
                  "system": system, "messages": [{"role": "user", "content": user}]})
    raw = "".join(b.get("text","") for b in resp.json().get("content",[])).replace("```json","").replace("```","").strip()
    try:
        return json.loads(raw)
    except Exception:
        raise HTTPException(500, f"AI returned invalid JSON: {raw[:200]}")

@app.post("/api/render")
async def render_reel(req: RenderRequest, background_tasks: BackgroundTasks):
    job_id = req.job_id or str(uuid.uuid4())
    job_dir = OUTPUT_DIR / job_id
    job_dir.mkdir(exist_ok=True)
    status_file = job_dir / "status.json"
    status_file.write_text(json.dumps({"status": "queued", "progress": 0, "message": "Starting…"}))
    background_tasks.add_task(run_render, req, job_dir, status_file)
    return {"job_id": job_id, "status": "queued"}

@app.get("/api/render/{job_id}/status")
def render_status(job_id: str):
    sf = OUTPUT_DIR / job_id / "status.json"
    if not sf.exists():
        raise HTTPException(404, "Job not found")
    return json.loads(sf.read_text())

@app.get("/api/render/{job_id}/download")
def download_render(job_id: str):
    f = OUTPUT_DIR / job_id / "reel.mp4"
    if not f.exists():
        raise HTTPException(404, "Render not ready")
    return FileResponse(str(f), media_type="video/mp4", filename="millican-pecan-reel.mp4")

def ws(sf, status, progress, message=""):
    sf.write_text(json.dumps({"status": status, "progress": progress, "message": message}))

def run_render(req: RenderRequest, job_dir: Path, status_file: Path):
    try:
        local_clips = [c.local_path for c in req.clips]
        inputs = []
        for p in local_clips:
            inputs += ["-i", p]

        ws(status_file, "processing", 20, "Preparing clips…")

        fparts = []
        for i in range(len(local_clips)):
            fparts.append(f"[{i}:v]scale=1080:1920:force_original_aspect_ratio=decrease,"
                          f"pad=1080:1920:(ow-iw)/2:(oh-ih)/2:black,setsar=1[v{i}];")

        ci = "".join(f"[v{i}][{i}:a]" for i in range(len(local_clips)))
        fparts.append(f"{ci}concat=n={len(local_clips)}:v=1:a=1[vconcat][aout];")

        font = "/usr/share/fonts/truetype/liberation/LiberationSans-Bold.ttf"

        def safe(t):
            return t.replace("'", "\u2019").replace(":", r"\:")

        fparts.append(f"[vconcat]drawtext=text='{safe(req.hook)}':fontfile={font}:"
                      f"fontsize=64:fontcolor=white:x=(w-text_w)/2:y=h*0.10:"
                      f"box=1:boxcolor=black@0.55:boxborderw=20:enable='between(t,0,3)'[vhook];")

        durations = []
        for p in local_clips:
            try:
                r = subprocess.run(["ffprobe","-v","error","-show_entries","format=duration",
                                    "-of","default=noprint_wrappers=1:nokey=1",p],
                                   capture_output=True, text=True, timeout=10)
                durations.append(float(r.stdout.strip()))
            except Exception:
                durations.append(8.0)

        prev = "[vhook]"
        t = 0.0
        for i, cap in enumerate(req.captions):
            if i >= len(local_clips):
                break
            s, e = t, t + durations[i] - 0.2
            t += durations[i]
            nl = f"[vcap{i}]"
            fparts.append(f"{prev}drawtext=text='{safe(cap)}':fontfile={font}:"
                          f"fontsize=52:fontcolor=white:x=(w-text_w)/2:y=h*0.82:"
                          f"box=1:boxcolor=black@0.50:boxborderw=16:"
                          f"enable='between(t,{s:.2f},{e:.2f})'{nl};")
            prev = nl

        cta_start = max(0, sum(durations) - 4)
        fparts.append(f"{prev}drawtext=text='{safe(req.cta)}':fontfile={font}:"
                      f"fontsize=60:fontcolor=white:x=(w-text_w)/2:y=h*0.88:"
                      f"box=1:boxcolor=black@0.65:boxborderw=20:"
                      f"enable='gte(t,{cta_start:.2f})'[vfinal]")

        ws(status_file, "rendering", 50, "FFmpeg rendering — takes a few minutes…")

        cmd = (["ffmpeg","-y"] + inputs +
               ["-filter_complex", "\n".join(fparts),
                "-map","[vfinal]","-map","[aout]",
                "-c:v","libx264","-preset","fast","-crf","23",
                "-c:a","aac","-b:a","128k","-movflags","+faststart",
                str(job_dir / "reel.mp4")])

        r = subprocess.run(cmd, capture_output=True, text=True, timeout=600)
        if r.returncode != 0:
            ws(status_file, "error", 0, f"FFmpeg error: {r.stderr[-600:]}")
            return
        ws(status_file, "done", 100, "Your reel is ready to download!")
    except Exception as e:
        ws(status_file, "error", 0, str(e))

frontend_path = Path(__file__).parent.parent / "frontend" / "dist"
if frontend_path.exists():
    app.mount("/", StaticFiles(directory=str(frontend_path), html=True), name="static")
